rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function uye() { return request.auth != null && request.auth.token.email in ['bus@queengames.app', 'yec@queengames.app']; }

    match /oyuncular/{uid} {
      allow read: if uye();
      allow write: if uye() && request.auth.uid == uid;
    }

    match /sohbet/{uid} {
      allow read: if uye();
      allow write: if uye() && request.auth.uid == uid && request.resource.data.size() <= 6 && request.resource.data.e is string && request.resource.data.e.size() <= 16 && (!('y' in request.resource.data) || (request.resource.data.y is string && request.resource.data.y.size() <= 60));
    }

    match /akis/{id} {
      allow read: if uye();
      allow create: if uye() && request.resource.data.uid == request.auth.uid && request.resource.data.size() <= 6;
      allow update, delete: if false;
    }

    match /hafiza/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.sira == request.auth.uid || resource.data.bitti == true || request.resource.data.diff(resource.data).affectedKeys().hasOnly(['gorulen', 't']));
      allow delete: if false;
    }

    match /satranc/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.sira == request.auth.uid || resource.data.bitti == true || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bitti', 'sonuc', 'kazanan', 'teklif', 't', 'v']) && request.resource.data.kazanan != request.auth.uid));
      allow delete: if false;
    }

    match /sayi/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.sira == request.auth.uid || resource.data.bitti == true
        || (resource.data.sira == "" && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['gizli', 'hazir', 'sira', 't', 'v']))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bitti', 'sonuc', 'kazanan', 'sira', 't', 'v']) && request.resource.data.kazanan != request.auth.uid));
      allow delete: if false;
    }

    match /amiral/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.sira == request.auth.uid || resource.data.bitti == true
        || (resource.data.sira == "" && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['yer', 'hazir', 'sira', 't', 'v']))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bitti', 'sonuc', 'kazanan', 'sira', 't', 'v']) && request.resource.data.kazanan != request.auth.uid));
      allow delete: if false;
    }

    match /asmaca/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.bitti == true || !('ilerleme' in resource.data)
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['ilerleme', 'yazilan', 'yazIp', 'hazir', 'bitti', 'sonuc', 'kazanan', 'sira', 't', 'v'])
          && request.resource.data.ilerleme.diff(resource.data.ilerleme).affectedKeys().hasOnly([request.auth.uid])
          && (!('yazilan' in request.resource.data.diff(resource.data).affectedKeys()) || request.resource.data.yazilan.diff(resource.data.yazilan).affectedKeys().hasOnly([request.auth.uid]))
          && (!('yazIp' in request.resource.data.diff(resource.data).affectedKeys()) || request.resource.data.yazIp.diff(resource.data.yazIp).affectedKeys().hasOnly([request.auth.uid]))
          && (!('hazir' in request.resource.data.diff(resource.data).affectedKeys()) || request.resource.data.hazir.diff(resource.data.hazir).affectedKeys().hasOnly([request.auth.uid]))));
      allow delete: if false;
    }

    match /xox/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.sira == request.auth.uid || resource.data.bitti == true
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bitti', 'sonuc', 'kazanan', 'sira', 'cizgi', 't', 'v']) && request.resource.data.kazanan != request.auth.uid));
      allow delete: if false;
    }

    match /tepe/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.sira == request.auth.uid || resource.data.bitti == true
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bitti', 'sonuc', 'kazanan', 'sira', 't', 'v']) && request.resource.data.kazanan != request.auth.uid));
      allow delete: if false;
    }

    match /zincir/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.sira == request.auth.uid || resource.data.bitti == true
        || (resource.data.mod == "yaz" && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['yer', 'hazir', 'cz', 'sira', 'puan', 'son', 'bitti', 'sonuc', 'kazanan', 't', 'v']))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bitti', 'sonuc', 'kazanan', 'sira', 't', 'v']) && request.resource.data.kazanan != request.auth.uid));
      allow delete: if false;
    }

    match /uno/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.sira == request.auth.uid || resource.data.sira == "bot" || resource.data.bitti == true
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bitti', 'sonuc', 'kazanan', 'sira', 't', 'v']) && request.resource.data.kazanan != request.auth.uid));
      allow delete: if false;
    }

    match /dort/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.sira == request.auth.uid || resource.data.bitti == true
        || request.resource.data.diff(resource.data).affectedKeys().hasOnly(['odul', 't'])
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bitti', 'sonuc', 'kazanan', 'sira', 'cizgi', 't', 'v']) && request.resource.data.kazanan != request.auth.uid));
      allow delete: if false;
    }

    match /kelimeavi/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye() && (id == "seri" || resource.data.bitti == true
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['oy', 'bitti', 'sonuc', 'kazanan', 't', 'v'])
          && request.resource.data.oy.diff(resource.data.oy).affectedKeys().hasOnly([request.auth.uid])));
      allow delete: if false;
    }

    match /ciz/{id} {
      allow read: if uye();
      allow create: if uye();
      allow update: if uye();
      allow delete: if false;
    }

    match /joled/{id} {
      allow read: if uye();
      allow create, update: if uye();
      allow delete: if false;
    }

    match /ciftlik/{uid} {
      allow read: if uye();
      allow write: if uye() && request.auth.uid == uid;
    }

    match /takvim/{uid} {
      allow read: if uye();
      allow write: if uye() && request.auth.uid == uid;
    }

    match /notlar/{id} {
      allow read: if uye();
      allow create: if uye() && request.resource.data.kimden == request.auth.uid
        && request.resource.data.keys().hasOnly(['kimden', 'kime', 'metin', 'tema', 't', 'acilis', 'okundu'])
        && request.resource.data.metin is string && request.resource.data.metin.size() > 0 && request.resource.data.metin.size() <= 300
        && request.resource.data.acilis is number && request.resource.data.okundu == null;
      allow update: if uye() && resource.data.kime == request.auth.uid && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['okundu']);
      allow delete: if uye() && resource.data.kimden == request.auth.uid && resource.data.okundu == null;
    }

    match /kediler/{id} {
      allow read: if uye();
      allow create: if uye() && request.resource.data.keys().hasOnly(['durum', 'isteyen', 'istendi', 'gun'])
        && request.resource.data.durum == 'bekliyor' && request.resource.data.isteyen == request.auth.uid;
      allow update: if uye() && (
        (resource.data.durum == 'bekliyor' && request.resource.data.durum == 'doydu' && request.resource.data.besleyen == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['durum', 'besleyen', 'doydu', 'foto', 'yer', 'ad']))
        || (resource.data.durum == 'doydu' && resource.data.besleyen == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['foto', 'yer', 'ad']))
        || (resource.data.isteyen == request.auth.uid && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['gordu'])));
      allow delete: if false;
    }

    match /kedifoto/{id} {
      allow read: if uye();
      allow create, update: if uye() && request.resource.data.keys().hasOnly(['v', 't'])
        && request.resource.data.v is string && request.resource.data.v.size() < 900000;
      allow delete: if false;
    }

    match /{rest=**} { allow read, write: if false; }
  }
}
